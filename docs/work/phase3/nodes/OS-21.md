  base_module() {
    export GOPATH="$PWD/.gopath" GOMODCACHE="$PWD/.gopath/pkg/mod" GOCACHE="$PWD/.gocache"
    printf 'module example.com/gotestfix\n\ngo 1.20\n' > go.mod

  }
